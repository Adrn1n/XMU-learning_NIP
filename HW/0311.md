## (2)
$$\begin{align}
\frac{1.5\text{MB}}{1\text{KB}} & = \frac{1.5 \times 2^{10}}{1} \\
& = 1.5 \times 2^{10} \\
\therefore t & = 2 \times \text{RTT} + \frac{\text{RTT}}{2} + (1.5 \times 2^{10} - 1) \text{RTT} + \frac{1.5\text{MB}}{10\text{Mbit/s}} \\
& = 160\text{ms} + \frac{80}{2}\text{ms} + (1.5 \times 2^{10} - 1) \times 80\text{ms} + \frac{1.5 \times 2^{20} \times 8}{10 \times 10^{20}}\text{s} \\
& = \frac{9707679}{78125}\text{s} \\
\end{align}$$

## (3)
$$\begin{align}
\frac{1.5\text{MB}}{1\text{KB} \times 20} & = \frac{1.5 \times 2^{10}}{1 \times 20} \\
& = \frac{384}{5} \\
\therefore t & = 2 \times \text{RTT} + \frac{\text{RTT}}{2} + (77 - 1) \times \text{RTT} \\
& = 160\text{ms} + \frac{80}{2}\text{ms} + (77 - 1) \times 80\text{ms} \\
& = 6.28\text{s} \\
\end{align}$$

## (4)
$$\begin{align}
s_n & = \frac{2^{n} - 1}{2 - 1} \times 1 \\
& = 2^n - 1 \\
2^n - 1 & \ge 1.5 \times 2^{10} \\
2^n & \ge 1.5 \times 2^{10} - 1 \\
n & \ge \log{(1.5 \times 2^{10} - 1)} \\
\therefore n & \ge 11 \\
\therefore t & = 2 \times \text{RTT} + \frac{\text{RTT}}{2} + (11 - 1) \times \text{RTT} \\
& = 160\text{ms} + \frac{80}{2}\text{ms} + (11 - 1) \times 80\text{ms} \\
& = 1\text{s} \\
\end{align}$$
